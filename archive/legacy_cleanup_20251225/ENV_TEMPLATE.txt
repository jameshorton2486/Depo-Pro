# ───────────────────────────────────────────────────────────────
# DATABASE (Supabase PostgreSQL)
# ───────────────────────────────────────────────────────────────
# Get from: Supabase → Settings → Database → Connection string (URI)
# Format: postgresql://postgres:PASSWORD@db.xxxxx.supabase.co:5432/postgres
DATABASE_URL=postgresql://postgres:YOUR_PASSWORD@db.YOUR_PROJECT.supabase.co:5432/postgres

# ───────────────────────────────────────────────────────────────
# SUPABASE API KEYS
# ───────────────────────────────────────────────────────────────
# Get from: Supabase → Settings → API

# Project URL (public - safe to expose in frontend)
NEXT_PUBLIC_SUPABASE_URL=https://YOUR_PROJECT.supabase.co

# Anon/Public Key (public - safe to expose in frontend)
# Used for client-side operations with RLS protection
NEXT_PUBLIC_SUPABASE_ANON_KEY=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.YOUR_ANON_KEY

# Service Role Key (⚠️ KEEP SECRET! Never expose in frontend)
# Used for server-side admin operations, bypasses RLS
SUPABASE_SERVICE_ROLE_KEY=eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.YOUR_SERVICE_ROLE_KEY

# ───────────────────────────────────────────────────────────────
# FILE STORAGE (Supabase Storage)
# ───────────────────────────────────────────────────────────────
# Enable cloud storage for file uploads
USE_CLOUD_STORAGE=true

# Bucket name (must match bucket created in Supabase Storage)
# Create at: Supabase → Storage → Create bucket
SUPABASE_STORAGE_BUCKET=depo-files

# ───────────────────────────────────────────────────────────────
# AUTHENTICATION (NextAuth.js v5)
# ───────────────────────────────────────────────────────────────
# Generate with: openssl rand -base64 32
# Or visit: https://generate-secret.vercel.app/32
NEXTAUTH_SECRET=YOUR_RANDOM_32_CHAR_SECRET_HERE

# Your application URL (update after Vercel deployment)
# Local: http://localhost:3000
# Production: https://your-app-name.vercel.app
NEXTAUTH_URL=http://localhost:3000

# ───────────────────────────────────────────────────────────────

# Get App Password from: https://myaccount.google.com/apppasswords
# (Requires 2-Step Verification enabled)

SMTP_PORT=587
SMTP_USER=your-email@gmail.com
